// Battle settings of the local server that differ from rAthena's defaults (conf/battle/*.conf).

// A pet fights beside its master: it attacks what the master attacks, and what strikes the master.
// It does so only when loyal enough (pet_support_min_friendly, 900 of 1000 by default, left as is).
pet_attack_support: yes
pet_damage_support: yes
// The line about the server's version on entering the game is not shown.
display_version: no

// A monster's blow counts on the server at the moment the client shows it land (the frame of the monster's
// sprite that the blow is on), not at the end of the monster's swing. The one struck is stopped at that moment,
// so the client and the server stop it in the same place: without this a character struck while it runs is
// stopped later on the client's screen than on the server, and jumps back at its next step.
synchronize_damage: yes

// The classic game: a blacksmith NPC refines through its dialog, not through the Refine UI window of later
// clients (which this client does not have: without this the NPC says nothing).
feature.refineui: off
